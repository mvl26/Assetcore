4.1 Workflow Specification
Mỗi DocType có lifecycle ≥ 4 trạng thái dùng Frappe Workflow Engine. Workflow định nghĩa trong fixtures và immutable trên prod (chỉ thay đổi qua patch). Mỗi state có doc_status (0=Draft, 1=Submitted, 2=Cancelled), allow_edit_roles, transition rules với điều kiện guard.

# assetcore/asset_registry/doctype/medical_asset/medical_asset.py
def get_permission_query_conditions(user):
    if not user:
        user = frappe.session.user
    if "AssetCore HTM Manager" in frappe.get_roles(user) or            "AssetCore QMS Officer" in frappe.get_roles(user) or            "AssetCore Auditor" in frappe.get_roles(user):
        return ""  # full access at query level
    user_depts = get_user_departments(user)  # custom helper
    if not user_depts:
        return "1=0"  # no access
    placeholders = ",".join([frappe.db.escape(d) for d in user_depts])
    return f"\`tabMedical Asset\`.department in ({placeholders})"
def has_permission(doc, user=None, permission_type=None):
    if not user:
        user = frappe.session.user
    roles = frappe.get_roles(user)
    if "AssetCore HTM Manager" in roles or        "AssetCore QMS Officer" in roles or        "AssetCore Auditor" in roles:
        return True
    if "AssetCore Operator" in roles:
        return doc.department in get_user_departments(user) and                doc.state in ("released_for_use", "in_use", "idle")
    return doc.department in get_user_departments(user)

4.3.3 API gọi engine từ nơi khác
# assetcore/lifecycle/api.py
def emit(medical_asset, event_type, *, source_doctype, source_doc,
         payload=None, state_before=None, state_after=None,
         reason_code=None, actor=None):
    """Emit a lifecycle event. Idempotent if (source_doc, event_type) seen.
    """
    if frappe.db.exists("Lifecycle Event", {
        "medical_asset": medical_asset,
        "event_type": event_type,
        "source_doc": source_doc
    }):
        return  # already emitted (idempotent)
    doc = frappe.new_doc("Lifecycle Event")
    doc.medical_asset = medical_asset
    doc.event_type = event_type
    doc.event_at = frappe.utils.now()
    doc.actor = actor or frappe.session.user
    doc.source_doctype = source_doctype
    doc.source_doc = source_doc
    doc.payload_json = frappe.as_json(payload or {})
    doc.state_before = state_before
    doc.state_after = state_after
    doc.reason_code = reason_code
    doc.flags.ignore_permissions = True  # system event
    doc.insert()

def check_wo_sla():

    open_states = ("planned", "scheduled", "in_progress", "paused")

def escalate_notification(wo):
    overdue_hours = compute_overdue_hours(wo)
    level = "L0"
    if overdue_hours >= 1: level = "L1"
    if overdue_hours >= 4: level = "L2"
    if overdue_hours >= 24: level = "L3"

4.4.4 Downtime calculation
Downtime = thời gian asset ở state in_repair / out_of_service do WO này gây ra. Tính theo giờ vận hành (operating window) — mặc định 24/7, hoặc theo lịch khoa nếu khoa cấu hình giờ vận hành riêng.
def compute_downtime(wo):
    asset = wo.medical_asset
    op_window = get_operating_window(asset.department)  # 24/7 or 8/5...
    in_repair_intervals = get_state_intervals(asset, "in_repair", wo.actual_start, wo.actual_end)
    return sum(intersect(i, op_window) for i in in_repair_intervals)

4.5.2 Distribution & training
Distribution List liệt kê User/Role/Group cần biết tài liệu effective.
Khi MAJOR change, hệ thống gửi notification 'Required reading' và yêu cầu acknowledgement trong 14 ngày.
Nếu QMS Artifact bật `required_training=1`, sinh Task 'Re-training' cho từng user trong distribution list, hết 30 ngày tự đóng nếu chưa hoàn thành thì cảnh báo QMS.
Acknowledgement của user lưu vào child Document Distribution với timestamp.
4.5.3 Retention engine
Mỗi Document Type có `retention_years` chuẩn (theo bảng 3.5.4).
Sau khi document chuyển 'retired' hoặc 'expired', timer retention bắt đầu chạy từ effective_until (hoặc retired_at).
Khi retention hết, hệ thống đánh dấu document 'archived' và move file ra cold storage.
Cấm xoá document trước khi retention hết, kể cả admin.
4.5.4 Search & retrieval
Document Record được full-text-search trong Frappe global search. Bổ sung filter cluster: by document_type, by scope, by status, by effective_from year. Mỗi Medical Asset có panel 'Documents' show toàn bộ document liên kết qua Compliance Record.

4.6.3 CAPA effectiveness check
Effectiveness check là bước bắt buộc, không bỏ qua. Đo bằng tiêu chí định lượng (vd: số lần tái xảy ra trong 90 ngày = 0) hoặc định tính (vd: SOP đã được phổ biến, training acknowledged 100%).
# Pseudo-code effectiveness check
def evaluate_capa(capa):
    if capa.effectiveness_check.method == "incident_count":
        threshold = capa.effectiveness_check.threshold or 0
        actual = count_incidents_since(capa.closed_at,
                                       window_days=90,
                                       scope=capa.scope_filter)
        return actual <= threshold
    elif capa.effectiveness_check.method == "training_completion":
        return training_completion_rate(capa.training_target) >= 0.95
    elif capa.effectiveness_check.method == "manual":
        return capa.effectiveness_check.conclusion == "effective"

4.6.5 Audit log hash chain
Asset Audit Log dùng hash chain để phát hiện chỉnh sửa lén. Mỗi entry có entry_hash = SHA-256 (chain_index || prev_hash || payload). Khi verify, scan toàn bộ chain so sánh hash từng entry với prev_hash của entry tiếp theo.
# assetcore/audit/chain.py
def insert_audit_entry(payload):
    last = frappe.db.get_value("Asset Audit Log",
        {}, "name, chain_index, entry_hash",
        order_by="chain_index desc")
    chain_index = (last.chain_index + 1) if last else 1
    prev_hash = last.entry_hash if last else "0" * 64
    payload_str = canonical_json(payload)
    entry_hash = sha256(f"{chain_index}|{prev_hash}|{payload_str}").hexdigest()
    doc = frappe.new_doc("Asset Audit Log")
    for k, v in payload.items():
        setattr(doc, k, v)
    doc.chain_index = chain_index
    doc.prev_hash = prev_hash
    doc.entry_hash = entry_hash
    doc.flags.ignore_permissions = True
    doc.insert()
def verify_audit_chain(start=1, end=None):
    rows = get_audit_rows(start, end)
    prev_hash = rows[0].prev_hash
    for r in rows:
        expected = sha256(f"{r.chain_index}|{prev_hash}|{canonical_json(r.payload())}").hexdigest()
        if expected != r.entry_hash:
            return False, r.chain_index
        prev_hash = r.entry_hash
    return True, None
Verify audit chain chạy nightly và export báo cáo lên dashboard QMS. Nếu phát hiện lệch, dispatch alert P1 đến QMS Officer và IT Admin.

@frappe.whitelist()
def snapshot_metric(metric_code, scope_doctype=None, scope_doc=None):
